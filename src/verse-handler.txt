let pinnedCite=null,hoverTimer=null,scrolling=null;
  const pop=(()=>{let el=document.getElementById("versepop");
    if(!el){el=document.createElement("div");el.id="versepop";el.className="versepop";document.body.appendChild(el);}
    return el;})();
  const hidePop=()=>{pop.classList.remove("on");};
  const clearVerse=()=>{pinnedCite=null;const b=$("#versebox");if(b)b.innerHTML="";hidePop();
    document.querySelectorAll(".chip.open,.chip.peek").forEach(x=>x.classList.remove("open","peek"));};
  const buildHtml=(cite,v,side,closeBtn)=>{
    if(!v||!v.verses)
      return '<div class="versebox vmiss"><div class="vr"><span>'+esc(cite)+'</span>'+(closeBtn?'<button data-close>✕ close</button>':'')+'</div><div class="vwhy">This reference is cited by the claim but isn\'t in the built-in reader — open it in your own Bible.</div></div>';
    const why=findWhy(d,cite);
    const whyHtml=why?"<b>Why it's cited:</b> "+esc(why):ROLENOTE(side);
    const anyHl=v.verses.some(x=>x.hl);
    const hasCtx=anyHl&&v.verses.some(x=>!x.hl);
    const ctx=hasCtx?' <span class="vcited">— '+esc(v.ref)+', with the verses either side</span>':"";
    const body=v.verses.map(x=>{
      const lit=(x.hl||!anyHl);
      const txt=(x.hl&&hasCtx)?"<em>"+esc(x.t)+"</em>":esc(x.t);
      return '<span class="vs'+(lit?" hl":"")+'"><span class="vnum">'+x.v+'</span>'+txt+' </span>';
    }).join("");
    return '<div class="versebox"><div class="vr"><span>'+esc(cite)+ctx+'</span>'+(closeBtn?'<button data-close>✕ close</button>':'')+'</div><div class="vtext">'+body+'</div><div class="vwhy">'+whyHtml+'</div></div>';
  };
  const sideOf=ch=>ch.closest("[data-side]")?ch.closest("[data-side]").dataset.side:"";
  /* hover = floating card, so nothing on the page moves */
  const peek=ch=>{
    const cite=ch.dataset.cite;
    pop.innerHTML=buildHtml(cite,VERSES[cite],sideOf(ch),false);
    pop.classList.add("on");
    const r=ch.getBoundingClientRect();
    const w=Math.min(560,window.innerWidth-24);
    pop.style.width=w+"px";
    let left=Math.min(Math.max(12,r.left),window.innerWidth-w-12);
    pop.style.left=left+"px";
    const h=pop.offsetHeight;
    pop.style.top=(r.bottom+8+h>window.innerHeight-8?Math.max(8,r.top-h-8):r.bottom+8)+"px";
    document.querySelectorAll(".chip.peek").forEach(x=>x.classList.remove("peek"));
    ch.classList.add("peek");
  };
  /* click = pinned in the reading column, beside what you clicked */
  const pinVerse=ch=>{
    hidePop();
    const cite=ch.dataset.cite,box=$("#versebox");
    if(!box)return;
    const host=ch.closest(".pblk")||ch.closest(".claim")||null;
    if(host&&host.parentNode)host.parentNode.insertBefore(box,host.nextSibling);
    document.querySelectorAll(".chip.open,.chip.peek").forEach(x=>x.classList.remove("open","peek"));
    ch.classList.add("open");pinnedCite=cite;
    box.innerHTML=buildHtml(cite,VERSES[cite],sideOf(ch),true);
    const cb=box.querySelector("[data-close]");if(cb)cb.addEventListener("click",clearVerse);
  };
  window.addEventListener("scroll",()=>{clearTimeout(scrolling);hidePop();
    scrolling=setTimeout(()=>{scrolling=null;},180);},{passive:true});
  const canHover=window.matchMedia("(hover:hover)").matches;
  document.querySelectorAll(".chip").forEach(ch=>{
    ch.addEventListener("click",()=>{
      if(ch.dataset.ext){window.open(ch.dataset.ext,"_blank");return;}
      if(pinnedCite===ch.dataset.cite){clearVerse();return;}
      pinVerse(ch);
    });
    if(!ch.dataset.ext&&canHover){
      ch.addEventListener("mouseenter",()=>{clearTimeout(hoverTimer);
        hoverTimer=setTimeout(()=>{if(!scrolling)peek(ch);},260);});
      ch.addEventListener("mouseleave",()=>{clearTimeout(hoverTimer);
        hoverTimer=setTimeout(()=>{hidePop();ch.classList.remove("peek");},120);});
    }
  });
