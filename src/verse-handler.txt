let pinnedCite=null,hoverTimer=null,scrolling=false,scrollT=null;
  const vbox=()=>$("#versebox");
  const clearVerse=()=>{pinnedCite=null;const b=vbox();if(b)b.innerHTML="";
    document.querySelectorAll(".chip.open,.chip.peek").forEach(x=>x.classList.remove("open","peek"));};
  const sideOf=ch=>ch.closest("[data-side]")?ch.closest("[data-side]").dataset.side:"";
  const buildHtml=(cite,v,side)=>{
    if(!v||!v.verses)
      return '<div class="versebox vmiss"><div class="vr"><span>'+esc(cite)+'</span><button data-close>✕ close</button></div><div class="vwhy">This reference is cited by the claim but isn\'t in the built-in reader — open it in your own Bible.</div></div>';
    const why=findWhy(d,cite);
    const whyHtml=why?"<b>Why it's cited:</b> "+esc(why):ROLENOTE(side);
    const anyHl=v.verses.some(x=>x.hl);
    const hasCtx=anyHl&&v.verses.some(x=>!x.hl);
    const ctx=hasCtx?' <span class="vcited">— '+esc(v.ref)+', with the verses either side</span>'
      :(!anyHl?' <span class="vcited">— '+esc(v.ref)+'</span>':'');
    const body=v.verses.map(x=>{
      const lit=(x.hl||!anyHl);
      const txt=(x.hl&&hasCtx)?"<em>"+esc(x.t)+"</em>":esc(x.t);
      return '<span class="vs'+(lit?" hl":"")+'"><span class="vnum">'+x.v+'</span>'+txt+' </span>';
    }).join("");
    return '<div class="versebox"><div class="vr"><span>'+esc(cite)+ctx+'</span><button data-close>✕ close</button></div><div class="vtext">'+body+'</div><div class="vwhy">'+whyHtml+'</div></div>';
  };
  /* The box opens in place, beside what you pointed at. The old glitch came from it
     opening while the page was scrolling, so the page kept resizing under the cursor.
     Fix: a hover delay, and nothing opens while a scroll is in progress. */
  const openVerse=(ch,pin)=>{
    const box=vbox(); if(!box)return;
    const cite=ch.dataset.cite;
    const host=ch.closest(".pblk, .claim, .defense, .talk, .vbox, .why, .saybox, li, p")
      ||ch.closest(".chips") && ch.closest(".claimside");
    if(host&&host.parentNode&&box.previousElementSibling!==host)host.parentNode.insertBefore(box,host.nextSibling);
    document.querySelectorAll(".chip.open,.chip.peek").forEach(x=>x.classList.remove("open","peek"));
    ch.classList.add(pin?"open":"peek");
    if(pin)pinnedCite=cite;
    box.innerHTML=buildHtml(cite,VERSES[cite],sideOf(ch));
    const cb=box.querySelector("[data-close]");if(cb)cb.addEventListener("click",clearVerse);
    /* the cited verse can sit below three verses of context — bring it into view */
    const inner=box.querySelector(".versebox"),hl=box.querySelector(".vs.hl");
    if(inner&&hl){
      const want=hl.offsetTop-inner.offsetTop-Math.max(0,(inner.clientHeight-hl.offsetHeight)/2);
      inner.scrollTop=Math.max(0,want);
    }
  };
  window.addEventListener("scroll",()=>{scrolling=true;clearTimeout(scrollT);clearTimeout(hoverTimer);
    scrollT=setTimeout(()=>{scrolling=false;},220);},{passive:true});
  /* LIVEREFS: references printed inside prose become openable, like the chips */
  document.querySelectorAll("#main .ref-bib, #main .ref-lds").forEach(r=>{
    const key=r.textContent.replace(/\s+/g," ").trim().replace(/[.,;:]$/,"");
    const hit=VERSES[key]?key:Object.keys(VERSES).find(k=>k.replace(/[\u2013\u2014]/g,"-")===key.replace(/[\u2013\u2014]/g,"-"));
    if(!hit)return;
    r.classList.add("live","chip");r.dataset.cite=hit;
  });
  const canHover=window.matchMedia("(hover:hover)").matches;
  document.querySelectorAll(".chip").forEach(ch=>{
    ch.addEventListener("click",()=>{
      if(ch.dataset.ext){window.open(ch.dataset.ext,"_blank");return;}
      if(pinnedCite===ch.dataset.cite){clearVerse();return;}
      openVerse(ch,true);
    });
    if(!ch.dataset.ext&&canHover){
      ch.addEventListener("mouseenter",()=>{clearTimeout(hoverTimer);
        if(pinnedCite||scrolling)return;
        hoverTimer=setTimeout(()=>{if(!scrolling&&!pinnedCite)openVerse(ch,false);},240);});
      ch.addEventListener("mouseleave",()=>{clearTimeout(hoverTimer);
        if(pinnedCite)return;
        hoverTimer=setTimeout(()=>{if(!pinnedCite)clearVerse();},160);});
    }
  });
